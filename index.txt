      summarySection.classList.add('hidden');
      stepMood.classList.remove('hidden');
    });

    function calculatePeriodStats(targetDateStr) {
      const targetDate = parseDateIT(targetDateStr);
      
      const moodCountsMonth = {};
      const year = targetDate.getFullYear();
      const month = targetDate.getMonth();
      const daysInMonth = new Date(year, month + 1, 0).getDate();

      for (let d = 1; d <= daysInMonth; d++) {
        const k = formatDateIT(new Date(year, month, d));
        if (savedData[k] && MOODS.includes(savedData[k].mood)) {
          const mName = savedData[k].mood;
          moodCountsMonth[mName] = (moodCountsMonth[mName] || 0) + 1;
        }
      }

      const getDominant = (counts) => {
        let dominant = null;
        let max = 0;
        for (const [m, c] of Object.entries(counts)) {
          if (c > max) { max = c; dominant = m; }
        }
        return dominant;
      };

      return {
        monthDominant: getDominant(moodCountsMonth)
      };
    }

    function showSummaryView(dateStr, data) {
      currentEditingDate = dateStr;
      
      document.getElementById('cardDate').innerText = dateStr === todayKey ? "OGGI - " + dateStr : dateStr;
      document.getElementById('cardWeather').innerText = data ? (data.weather || '') : '';
      
      const moodTag = document.getElementById('cardMoodTag');
      if (data && data.mood && MOODS.includes(data.mood)) {
        moodTag.innerText = data.mood;
        moodTag.style.background = data.color || (userColorMap ? userColorMap[data.mood] : '#333');
        moodTag.style.display = 'inline-block';
      } else {
        moodTag.style.display = 'none';
      }

      const cardSong = document.getElementById('cardSong');
      const songTitle = document.getElementById('cardSongTitle');
      const songLink = document.getElementById('cardSongLink');

      if (data && data.songData) {
        songTitle.innerText = data.songData.title;
        if (data.songData.url) {
          songLink.href = data.songData.url;
          songLink.classList.remove('hidden');
        } else {
          songLink.classList.add('hidden');
        }
        cardSong.classList.remove('hidden');
      } else {
        cardSong.classList.add('hidden');
      }

      const notesElem = document.getElementById('cardNotes');
      notesElem.innerText = data && data.notes ? data.notes : 'Nessuna nota presente per questo giorno.';
      notesElem.style.display = 'block';

      const photosContainer = document.getElementById('cardPhotos');
      photosContainer.innerHTML = '';
      if (data && data.photos && data.photos.length > 0) {
        photosContainer.style.display = 'grid';
        data.photos.forEach(src => {
          const img = document.createElement('img');
          img.src = src;
          photosContainer.appendChild(img);
        });
      } else {
        photosContainer.style.display = 'none';
      }

      const btnDelete = document.getElementById('btnDeleteCard');
      const btnEdit = document.getElementById('btnEditCard');

      if (!data) {
        btnDelete.style.display = 'none';
        btnEdit.innerText = "Compila questa giornata";
      } else {
        btnDelete.style.display = 'block';
        btnEdit.innerText = "Modifica questa giornata";
      }

      updateStreakDisplays();
      renderCarousel();
    }

    function renderCarousel() {
      const carousel = document.getElementById('rotationCarousel');
      carousel.innerHTML = '';
      
      const currentYear = new Date().getFullYear();
      const startDate = new Date(currentYear, 0, 1);
      
      const todayDate = new Date();
      todayDate.setHours(0,0,0,0);

      for (let i = 0; i < 365; i++) {
        const currentDate = new Date(startDate);
        currentDate.setDate(startDate.getDate() + i);
        const dateKey = formatDateIT(currentDate);
        
        const item = document.createElement('div');
        item.className = 'carousel-item' + (dateKey === currentEditingDate ? ' active' : '');
        
        if (currentDate > todayDate) {
          item.classList.add('is-disabled');
        }

        if (savedData[dateKey] && MOODS.includes(savedData[dateKey].mood)) {
          const mColor = savedData[dateKey].color || (userColorMap ? userColorMap[savedData[dateKey].mood] : '#333');
          item.style.backgroundColor = mColor;
          item.style.borderColor = 'transparent';
          item.classList.add('has-mood');
        }

        const dayLabel = document.createElement('span');
        dayLabel.className = 'item-day';
        const dayNum = String(currentDate.getDate()).padStart(2, '0');
        const monthNum = String(currentDate.getMonth() + 1).padStart(2, '0');
        dayLabel.innerText = `${dayNum}/${monthNum}`;

        const yearLabel = document.createElement('span');
        yearLabel.className = 'item-year';
        yearLabel.innerText = currentDate.getFullYear();

        item.appendChild(dayLabel);
        item.appendChild(yearLabel);

        if (currentDate <= todayDate) {
          item.addEventListener('click', () => {
            if (savedData[dateKey]) {
              showSummaryView(dateKey, savedData[dateKey]);
            } else {
              currentEditingDate = dateKey;
              showSummaryView(dateKey, null);
            }
          });
        }

        carousel.appendChild(item);
      }

      setTimeout(() => {
        const activeItem = carousel.querySelector('.carousel-item.active');
        if (activeItem) {
          activeItem.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' });
        }
      }, 100);
    }

    function renderYearPalette() {
      const bar = document.getElementById('yearPaletteBar');
      const legend = document.getElementById('yearPaletteLegend');
      bar.innerHTML = '';
      legend.innerHTML = '';

      const moodCounts = {};
      let totalLogged = 0;
      const currentYear = new Date().getFullYear().toString();

      Object.entries(savedData).forEach(([dateKey, item]) => {
        if (dateKey.endsWith(currentYear) && item && item.mood && MOODS.includes(item.mood)) {
          moodCounts[item.mood] = (moodCounts[item.mood] || 0) + 1;
          totalLogged++;
        }
      });

      if (totalLogged === 0) {
        bar.innerHTML = `<div style="width:100%; height:100%; background:var(--bg-input);"></div>`;
        legend.innerHTML = `<span style="font-size:0.75rem; color:var(--text-muted);">Nessun dato registrato quest'anno.</span>`;
        return;
      }

      MOODS.forEach(mood => {
        const count = moodCounts[mood] || 0;
        if (count > 0) {
          const percentage = ((count / totalLogged) * 100).toFixed(1);
          const color = (userColorMap && userColorMap[mood]) || defaultColorMap[mood] || '#888';

          const segment = document.createElement('div');
          segment.className = 'palette-segment';
          segment.style.width = `${percentage}%`;
          segment.style.backgroundColor = color;
          segment.title = `${mood}: ${percentage}% (${count} giorni)`;
          bar.appendChild(segment);

          const legItem = document.createElement('div');
          legItem.className = 'legend-item';
          legItem.innerHTML = `<span class="legend-color" style="background:${color}"></span>${mood} <strong>${percentage}%</strong>`;
          legend.appendChild(legItem);
        }
      });
    }

    function renderFullGrid() {
      const container = document.getElementById('monthsContainer');
      container.innerHTML = '';

      const currentYear = new Date().getFullYear();
      document.getElementById('modalYearTitle').innerText = currentYear;

      const todayDate = new Date();
      todayDate.setHours(0,0,0,0);

      for (let m = 0; m < 12; m++) {
        const monthSection = document.createElement('div');
        monthSection.className = 'month-section';

        const monthHeaderFlex = document.createElement('div');
        monthHeaderFlex.className = 'month-header-flex';

        const monthTitle = document.createElement('div');
        monthTitle.className = 'month-title';
        monthTitle.innerText = MONTH_NAMES[m];
        monthHeaderFlex.appendChild(monthTitle);

        const statsContainer = document.createElement('div');
        statsContainer.className = 'month-stats-flex';

        const monthRefDate = formatDateIT(new Date(currentYear, m, 1));
        const periodStats = calculatePeriodStats(monthRefDate);

        if (periodStats.monthDominant) {
          const badgeMonth = document.createElement('div');
          badgeMonth.className = 'month-dominant-badge';
          const domColor = userColorMap ? userColorMap[periodStats.monthDominant] : '#333';
          badgeMonth.innerHTML = `<span class="month-dominant-dot" style="background:${domColor}"></span>Mese: ${periodStats.monthDominant}`;
          statsContainer.appendChild(badgeMonth);
        }

        monthHeaderFlex.appendChild(statsContainer);
        monthSection.appendChild(monthHeaderFlex);

        const monthGrid = document.createElement('div');
        monthGrid.className = 'month-grid';

        const daysInMonth = new Date(currentYear, m + 1, 0).getDate();

        for (let d = 1; d <= daysInMonth; d++) {
          const currentDate = new Date(currentYear, m, d);
          const dateKey = formatDateIT(currentDate);

          const square = document.createElement('div');
          square.className = 'day-square';

          if (currentDate > todayDate) {
            square.classList.add('is-disabled');
          }

          const dayStr = String(d).padStart(2, '0');
          const monthStr = String(m + 1).padStart(2, '0');

          const dateLabel = document.createElement('span');
          dateLabel.className = 'sq-date';
          dateLabel.innerText = `${dayStr}/${monthStr}`;
          square.appendChild(dateLabel);

          if (savedData[dateKey] && MOODS.includes(savedData[dateKey].mood)) {
            const mColor = savedData[dateKey].color || (userColorMap ? userColorMap[savedData[dateKey].mood] : '#333');
            square.style.backgroundColor = mColor;
            square.classList.add('has-data');
          }

          if (currentDate <= todayDate) {
            square.addEventListener('click', () => {
              if (savedData[dateKey]) {
                showSummaryView(dateKey, savedData[dateKey]);
              } else {
                currentEditingDate = dateKey;
                showSummaryView(dateKey, null);
              }
              gridModal.classList.add('hidden');
            });
          }

          monthGrid.appendChild(square);
        }

        monthSection.appendChild(monthGrid);
        container.appendChild(monthSection);
      }
    }
